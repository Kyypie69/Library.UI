-- Countdown: 3 seconds for smaller GUI
task.delay(3, function()
    waitBtn.Visible = false
    exitBtn.Visible = true
end)

copyBtn.MouseButton1Click:Connect(function()
    if setclipboard then
        pcall(function()
            setclipboard(discordLink)
        end)
        copyBtn.Text = "Copied!"
        copyBtn.BackgroundColor3 = Color3.fromRGB(100, 255, 100)
        task.wait(0.8)
        copyBtn.Text = "Copy Link"
        copyBtn.BackgroundColor3 = Color3.fromRGB(88, 101, 242)
    else
        fallbackBox.Text = discordLink
        fallbackBox.Visible = true
        pcall(function() fallbackBox:CaptureFocus() end)
    end
end)

exitBtn.MouseButton1Click:Connect(function()
    -- Fade out effect
    for i = 1, 10 do
        frame.BackgroundTransparency = 0.4 + (i * 0.06)
        outerGlow.Transparency = 0.85 + (i * 0.015)
        task.wait(0.02)
    end
    screenGui:Destroy()
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.Escape then
        screenGui.Enabled = false
    end
end)

task.spawn(function()

    end)
end)
